    USERS {
        int user_id PK
        string email UK
        string password_hash
        string role "patient | doctor | staff | admin"
        timestamp created_at
        timestamp last_login
    }
    CLINIC {
        int clinic_id PK
        string name
        string phone
        string email
        string address
        timestamp created_at
    }
    PATIENT {
        int patient_id PK
        int user_id FK "unique, nullable (walk-ins)"
        string first_name
        string last_name
        date date_of_birth
        string gender
        string phone
        string email
        string address
        string blood_type
        string allergies
        timestamp created_at
    }
    DOCTOR {
        int doctor_id PK
        int user_id FK "unique"
        int clinic_id FK
        string first_name
        string last_name
        string specialization
        string phone
        string license_number UK
        string bio
        timestamp created_at
    }
    DOCTOR_SCHEDULE {
        int schedule_id PK
        int doctor_id FK
        int day_of_week "0 = Sunday"
        time start_time
        time end_time
        int slot_minutes
    }
    STAFF {
        int staff_id PK
        int user_id FK "unique"
        int clinic_id FK
        string first_name
        string last_name
        string position "receptionist | nurse | ..."
        string phone
        timestamp created_at
    }
    APPOINTMENT {
        int appointment_id PK
        int patient_id FK
        int doctor_id FK
        date appointment_date
        time appointment_time
        string status "requested | scheduled | completed | cancelled | no_show"
        string reason
        string notes
        timestamp created_at
    }
    MEDICAL_RECORD {
        int record_id PK
        int patient_id FK
        int doctor_id FK

        string treatment_plan
        string notes
        timestamp created_at
    }

    }
    PRESCRIPTION {
        int prescription_id PK
        int record_id FK

        int refills_allowed
        string notes
        timestamp created_at
    }
    PRESCRIPTION_ITEM {
        int item_id PK

        string instructions
